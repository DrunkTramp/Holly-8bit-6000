# This repo is a package now: `pyproject.toml` holds the real dependency story.
#
#   .venv/bin/pip install -e .                 # runtime: numpy only, gives you `import holly`
#   .venv/bin/pip install -e .[build]          # + psd-tools/pillow to bake buffers from the PSD
#   .venv/bin/pip install -e .[player]         # + pygame for tools/reference_player.py
#
# The file below is kept so a bare `pip install -r requirements.txt` still reproduces the
# build environment on a fresh box. Versions verified on this machine (Python 3.12.3).

numpy==2.5.3
pillow==12.3.0
pygame==2.6.1
